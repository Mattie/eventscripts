cmake_minimum_required(VERSION 3.20)
project(EventScripts LANGUAGES NONE)

file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/build" ES_BUILD_ROOT)
file(REAL_PATH "${CMAKE_CURRENT_BINARY_DIR}" ES_BINARY_ROOT)
string(FIND "${ES_BINARY_ROOT}/" "${ES_BUILD_ROOT}/" ES_BUILD_PREFIX)
if(NOT ES_BUILD_PREFIX EQUAL 0 OR ES_BINARY_ROOT STREQUAL ES_BUILD_ROOT)
  message(FATAL_ERROR "Choose a binary directory beneath the project build directory, for example -B build/host.")
endif()

option(EVENTSCRIPTS_BUILD_NATIVE "Build standalone native compatibility components" ON)
option(EVENTSCRIPTS_BUILD_PLUGIN "Build the historical Windows x86 Orange Box plugin (VC++ 2005 SP1/Python 2.5)" OFF)
set(ES_TESTS_AVAILABLE OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
  set(ES_TESTS_AVAILABLE ON)
endif()
option(BUILD_TESTING "Build EventScripts tests" ${ES_TESTS_AVAILABLE})
if(BUILD_TESTING AND NOT ES_TESTS_AVAILABLE)
  message(FATAL_ERROR "BUILD_TESTING requires tests/CMakeLists.txt. Restore the tests tree or configure with -DBUILD_TESTING=OFF.")
endif()

if(EVENTSCRIPTS_BUILD_PLUGIN)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
if(EVENTSCRIPTS_BUILD_NATIVE OR EVENTSCRIPTS_BUILD_PLUGIN)
  enable_language(C CXX)
  add_subdirectory(src)
endif()

if(BUILD_TESTING)
  include(CTest)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_subdirectory(tests)
endif()

if(EVENTSCRIPTS_BUILD_PLUGIN)
  message(STATUS "mattie_eventscripts is a local historical-ABI rebuild; compile/link does not establish plugin loading or server compatibility.")
else()
  message(STATUS "Plugin build disabled; standalone native and host checks do not require the Source SDK or Python 2.5.")
endif()
